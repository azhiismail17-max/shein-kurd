function getOrCreateGiftCardSheet(ss) {
  var sheet = ss.getSheetByName(GIFT_CARD_SHEET_NAME);
  if (!sheet) sheet = ss.insertSheet(GIFT_CARD_SHEET_NAME);
  var headers = ["ID", "Date", "Card Number", "Card Pin", "Payment Method", "Payment Other", "Card Price", "Spent", "Remaining", "Linked Boxes", "Notes"];
  var existing = sheet.getRange(1, 1, 1, headers.length).getValues()[0];
  var needsHeader = false;
  for (var i = 0; i < headers.length; i++) {
    if (existing[i] !== headers[i]) { needsHeader = true; break; }
  }
  if (needsHeader) sheet.getRange(1, 1, 1, headers.length).setValues([headers]);
  return sheet;
}

function parseMoneyValue(value) {
  if (value === "" || value === null || value === undefined) return "";
  var parsed = parseFloat(value.toString().replace(/[^0-9.-]+/g, ""));
  return isNaN(parsed) ? 0 : parsed;
}

function parseGiftCardLinkedBoxes(value) {
  if (!value) return [];
  return value.toString().split(",").map(function(box) { return box.trim(); }).filter(function(box) { return box; });
}

function sortGiftCardsByBalance(cards) {
  return cards.map(function(card, index) {
    return { card: card, index: index };
  }).sort(function(a, b) {
    var aHasBalance = (parseFloat(a.card.remaining) || 0) > 0 ? 1 : 0;
    var bHasBalance = (parseFloat(b.card.remaining) || 0) > 0 ? 1 : 0;
    return (bHasBalance - aHasBalance) || (a.index - b.index);
  }).map(function(item) {
    return item.card;
  });
}

function readGiftCards(ss) {
  var sheet = getOrCreateGiftCardSheet(ss);
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return [];
  var values = sheet.getRange(2, 1, lastRow - 1, 11).getValues();
  var cards = [];
  for (var i = 0; i < values.length; i++) {
    var r = values[i];
    if (!r[0]) continue;
    cards.push({
      id: r[0].toString(),
      date: r[1] ? r[1].toString() : "",
      card_number: r[2] ? r[2].toString() : "",
      card_pin: r[3] ? r[3].toString() : "",
      payment_method: r[4] ? r[4].toString() : "",
      payment_other: r[5] ? r[5].toString() : "",
      card_price: parseFloat(r[6]) || 0,
      spent: parseFloat(r[7]) || 0,
      remaining: parseFloat(r[8]) || 0,
      linked_boxes: parseGiftCardLinkedBoxes(r[9]),
      notes: r[10] ? r[10].toString() : ""
    });
  }
  return sortGiftCardsByBalance(cards.reverse());
}

function findGiftCardRow(sheet, id) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return -1;
  var values = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
  for (var i = 0; i < values.length; i++) {
    if (values[i][0] && values[i][0].toString() === id.toString()) return i + 2;
  }
  return -1;
}

function spendGiftCard(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = getOrCreateGiftCardSheet(ss);
  var row = findGiftCardRow(sheet, data.id || "");
  if (row < 2) return { status: "error", message: "Gift card not found" };
  var amount = parseFloat(data.amount) || 0;
  if (amount === 0) return { status: "error", message: "Invalid amount" };
  var price = parseFloat(sheet.getRange(row, 7).getValue()) || 300;
  var spent = parseFloat(sheet.getRange(row, 8).getValue()) || 0;
  var nextSpent = Math.max(spent + amount, 0);
  var nextRemaining = Math.max(price - nextSpent, 0);
  sheet.getRange(row, 8).setValue(nextSpent);
  sheet.getRange(row, 9).setValue(nextRemaining);
  return { status: "success", spent: nextSpent, remaining: nextRemaining };
}

function linkGiftCardBox(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = getOrCreateGiftCardSheet(ss);
  var row = findGiftCardRow(sheet, data.id || "");
  if (row < 2) return { status: "error", message: "Gift card not found" };
  var boxName = (data.box_name || "").toString().trim();
  var sheetName = (data.sheet_name || data.sheet || getActiveSheetName()).toString().trim();

  var linked = parseGiftCardLinkedBoxes(sheet.getRange(row, 10).getValue());
  if (linked.indexOf(boxKey) === -1) linked.push(boxKey);
  sheet.getRange(row, 10).setValue(linked.join(","));
  return { status: "success", linked_boxes: linked };
}

function findBoxLinkRow(sheet, sheetName, boxName) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return -1;
  var values = sheet.getRange(2, 2, lastRow - 1, 2).getValues();
  for (var i = 0; i < values.length; i++) {
    if ((values[i][0] || "").toString() === sheetName && (values[i][1] || "").toString() === boxName) return i + 2;

function saveBoxLink(data) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = getOrCreateBoxLinkSheet(ss);
  var sheetName = (data.sheet_name || data.sheet || getActiveSheetName()).toString().trim();
  var boxName = (data.box_name || "").toString().trim();
  if (!boxName) return { status: "error", message: "Missing box name" };
  var totalLink = (data.total_link || data.totalLink || "").toString().trim();
  var imageUrl = (data.image_url || data.imageUrl || "").toString().trim();
  var updatedAt = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss");
  var updatedBy = (data.updated_by || data.admin_name || data.submitted_by || "").toString();
  var updatedRole = (data.role || data.updated_role || "").toString();
  var row = findBoxLinkRow(sheet, sheetName, boxName);
  var id = row > 1 ? sheet.getRange(row, 1).getValue().toString() : Utilities.getUuid();
  if (row < 2) {
    sheet.appendRow([id, sheetName, boxName, totalLink, "", imageUrl, updatedAt, updatedBy, updatedRole]);
    row = sheet.getLastRow();
  } else {
    sheet.getRange(row, 1, 1, 9).setValues([[id, sheetName, boxName, totalLink, "", imageUrl, updatedAt, updatedBy, updatedRole]]);
  }
  if (imageUrl) {
    sheet.getRange(row, 5).setFormula('=IMAGE("' + imageUrl.replace(/"/g, '""') + '")');
  } else {
    sheet.getRange(row, 5).clearContent();
  }
  return { status: "success", box_link: { id: id, sheet_name: sheetName, box_name: boxName, total_link: totalLink, image_url: imageUrl, updated_at: updatedAt, updated_by: updatedBy, updated_role: updatedRole }, box_links: readBoxLinks(ss) };
}

function handleBoxLinkAction(data) {
  var action = (data.action || "").toString().trim();
  if (["get_box_links", "save_box_link"].indexOf(action) === -1) return null;
  if (!canManageBoxLinkRole(data.role)) {
    return boxLinkJsonResponse({ status: "error", message: "Owner/Admin only" });
  }
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  if (action === "get_box_links") return boxLinkJsonResponse({ status: "success", box_links: readBoxLinks(ss) });
  return boxLinkJsonResponse(saveBoxLink(data));
}

function setActiveSheetName(sheetName) {
  var cleanSheetName = (sheetName || "").toString().trim();
  if (LOCAL_SHEETS.indexOf(cleanSheetName) === -1) {
    return { status: "error", message: "Invalid month: " + cleanSheetName };
  }

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var settings = ss.getSheetByName(SETTINGS_SHEET_NAME);
  if (!settings) {
    settings = ss.insertSheet(SETTINGS_SHEET_NAME);
    settings.getRange(1, 1, 1, 2).setValues([[ACTIVE_SHEET_SETTING_KEY, cleanSheetName]]);
    return { status: "success", active_sheet: cleanSheetName };
  }

  var lastRow = Math.max(settings.getLastRow(), 1);
  var values = settings.getRange(1, 1, lastRow, 2).getValues();
  for (var i = 0; i < values.length; i++) {
    if ((values[i][0] || "").toString().trim() === ACTIVE_SHEET_SETTING_KEY) {
      settings.getRange(i + 1, 2).setValue(cleanSheetName);
      return { status: "success", active_sheet: cleanSheetName };
    }
  }

  settings.getRange(lastRow + 1, 1, 1, 2).setValues([[ACTIVE_SHEET_SETTING_KEY, cleanSheetName]]);
  return { status: "success", active_sheet: cleanSheetName };
}

function parseLinkedRowIds(value, sheetName) {
  var input = value;
  if (input === null || input === undefined || input === "") return [];
  if (typeof input === "string") {
    var trimmed = input.trim();
    if (trimmed.charAt(0) === "[") {
      try { input = JSON.parse(trimmed); } catch(e) { input = trimmed.split(","); }
    } else {
      input = trimmed.split(",");
    }
  }
  if (!Array.isArray(input)) input = [input];

  var seen = {};
  var ids = [];
  for (var i = 0; i < input.length; i++) {
    var token = (input[i] || "").toString().trim();
    if (!token) continue;
    var rowPart = token;
    if (token.indexOf(":") > -1) {
      var parts = token.split(":");
      rowPart = parts[0];
      var tokenSheet = parts.slice(1).join(":").trim();
      if (tokenSheet && sheetName && tokenSheet !== sheetName) continue;
    }
    var rowId = parseInt(rowPart, 10);
    if (!isNaN(rowId) && rowId >= 4 && rowId <= MAX_ROW_LIMIT && !seen[rowId]) {
      seen[rowId] = true;
      ids.push(rowId);
    }
  }
  ids.sort(function(a, b) { return a - b; });
  return ids;
}

function parseLinkedCellTokens(value) {
  var input = value;
  if (input === null || input === undefined || input === "") return [];
  if (typeof input === "string") {
    var trimmed = input.trim();
    if (trimmed.charAt(0) === "[") {
      try { input = JSON.parse(trimmed); } catch(e) { input = trimmed.split(","); }
    } else {
      input = trimmed.split(",");
    }
  }
  if (!Array.isArray(input)) input = [input];

  var seen = {};
  var tokens = [];
  for (var i = 0; i < input.length; i++) {
    var token = (input[i] || "").toString().trim();
    if (!token || seen[token]) continue;
    seen[token] = true;
    tokens.push(token);
  }
  return tokens;
}

function normalizeOrderRef(rowId, sheetName) {
  return parseInt(rowId, 10) + ":" + sheetName;
}

function writeOrderLinkRefs(sheet, rowId, refs) {
  var seen = {};
  var cleanRefs = [];
  for (var i = 0; i < refs.length; i++) {
    var ref = (refs[i] || "").toString().trim();
    if (!ref || seen[ref]) continue;
    seen[ref] = true;
    cleanRefs.push(ref);
  }
  if (cleanRefs.length > 0) sheet.getRange(parseInt(rowId, 10), 28).setValue(cleanRefs.join(","));
  else sheet.getRange(parseInt(rowId, 10), 28).clearContent();
}

function parseOrderLinkRefs(value, fallbackSheetName) {
  var tokens = parseLinkedCellTokens(value || []);
  var refs = [];
  var seen = {};
  for (var i = 0; i < tokens.length; i++) {
    var token = (tokens[i] || "").toString().replace(/^!/, "").trim();
    if (!token) continue;
    var parts = token.split(":");
    var row = parseInt(parts[0], 10);
    var sheetName = parts.length > 1 ? parts.slice(1).join(":").trim() : fallbackSheetName;
    if (!row || row < 4 || row > MAX_ROW_LIMIT || !sheetName) continue;
    var key = normalizeOrderRef(row, sheetName);
    if (seen[key]) continue;
    seen[key] = true;
    refs.push({ row: row, sheetName: sheetName, key: key });
  }
  return refs;
}

function getCompleteLinkedRefs(seedRefs) {
  var queue = parseOrderLinkRefs(seedRefs || [], "");
  var seen = {};
  for (var i = 0; i < queue.length; i++) seen[queue[i].key] = true;

  var index = 0;
  while (index < queue.length) {
    var current = queue[index++];
    var sheet = getOrderSheetByName(current.sheetName);
    if (!sheet) continue;
    var linkedValue = "";
    try { linkedValue = sheet.getRange(current.row, 28).getValue().toString(); } catch(e) {}
    var linkedRefs = parseOrderLinkRefs(linkedValue, current.sheetName);
    for (var j = 0; j < linkedRefs.length; j++) {
      if (!seen[linkedRefs[j].key]) {
        seen[linkedRefs[j].key] = true;
        queue.push(linkedRefs[j]);
      }
    }
  }
  return queue;
}

function gsNormalizePhoneNumber(value) {
  var digits = (value || "").toString().replace(/\D/g, "");
  if (digits.indexOf("964") === 0) digits = digits.slice(3);
  if (digits.indexOf("0") === 0) digits = digits.slice(1);
  var m = digits.match(/7\d{9}/);
  if (!m) return "";
  var phone = m[0];
  if (/^(\d)\1+$/.test(phone)) return "";
  return phone;
}

function gsGetRowPhones(sheet, row) {
  var raw = sheet.getRange(row, 9).getValue().toString();
  var parts = raw.split("/");
  var phones = [];
  for (var i = 0; i < parts.length; i++) {
    var p = gsNormalizePhoneNumber(parts[i]);
    if (p && phones.indexOf(p) === -1) phones.push(p);
  }
  return phones;
}

function gsGetRowInstagram(sheet, row) {
  return sheet.getRange(row, 1).getValue().toString().trim().toLowerCase().replace(/^@+/, "").replace(/\s+/g, "");
}

// Same rule as the client (OrderFormView / OrderListView): two orders may
// only be linked if they share a phone number or an Instagram handle.
// Checked again here, at the point the sheet is actually written, so a bad
// link can never reach the data no matter which client sent it - the
// picker, the bulk-link action, or an old cached app version still running
// on someone's phone that predates a client-side fix.
function gsIsSameCustomerRow(sheetA, rowA, sheetB, rowB) {
  var instaA = gsGetRowInstagram(sheetA, rowA);
  var instaB = gsGetRowInstagram(sheetB, rowB);
  if (instaA && instaA === instaB) return true;
  var phonesA = gsGetRowPhones(sheetA, rowA);
  var phonesB = gsGetRowPhones(sheetB, rowB);
  for (var i = 0; i < phonesA.length; i++) {
    if (phonesB.indexOf(phonesA[i]) > -1) return true;
  }
  return false;
}

function relinkOrderRefs(targetRow, targetSheetName, linkedRefs) {
  var targetRef = normalizeOrderRef(targetRow, targetSheetName);
  var targetSheetObj = getOrderSheetByName(targetSheetName);
  var newGroup = parseOrderLinkRefs(linkedRefs || [], targetSheetName);
  if (targetSheetObj) {
    newGroup = newGroup.filter(function(ref) {
      if (ref.key === targetRef) return true;
      var refSheet = getOrderSheetByName(ref.sheetName);
      if (!refSheet) return false;
      return gsIsSameCustomerRow(targetSheetObj, targetRow, refSheet, ref.row);
    });
  }
  if (newGroup.filter(function(o) { return o.key === targetRef; }).length === 0) {
    newGroup.push({ row: parseInt(targetRow, 10), sheetName: targetSheetName, key: targetRef });
  }

  var newKeys = {};
  for (var i = 0; i < newGroup.length; i++) newKeys[newGroup[i].key] = true;

  var oldGroups = [];
  var affected = {};
  var seeds = newGroup.slice();
  seeds.push({ row: parseInt(targetRow, 10), sheetName: targetSheetName, key: targetRef });
  for (var s = 0; s < seeds.length; s++) {
    var group = getCompleteLinkedRefs([seeds[s].key]);
    oldGroups.push(group);
    for (var g = 0; g < group.length; g++) affected[group[g].key] = group[g];
  }
  for (var n = 0; n < newGroup.length; n++) affected[newGroup[n].key] = newGroup[n];

  for (var key in affected) {
    if (!affected.hasOwnProperty(key)) continue;
    var current = affected[key];
    var sheet = getOrderSheetByName(current.sheetName);
    if (sheet) writeOrderLinkRefs(sheet, current.row, []);
  }

  for (var og = 0; og < oldGroups.length; og++) {
    var remaining = oldGroups[og].filter(function(o) { return !newKeys[o.key]; });
    if (remaining.length < 2) continue;
    for (var r = 0; r < remaining.length; r++) {
      var rem = remaining[r];
      var remSheet = getOrderSheetByName(rem.sheetName);
      if (!remSheet) continue;
      writeOrderLinkRefs(remSheet, rem.row, remaining.filter(function(o) { return o.key !== rem.key; }).map(function(o) { return o.key; }));
    }
  }

  if (newGroup.length > 1) {
    for (var x = 0; x < newGroup.length; x++) {
      var order = newGroup[x];
      var orderSheet = getOrderSheetByName(order.sheetName);
      if (!orderSheet) continue;
      writeOrderLinkRefs(orderSheet, order.row, newGroup.filter(function(o) { return o.key !== order.key; }).map(function(o) { return o.key; }));
    }
  }
}

function getCompleteLinkedRows(sheet, seedRows, sheetName) {
  var queue = parseLinkedRowIds(seedRows, sheetName);
  var seen = {};
  for (var i = 0; i < queue.length; i++) seen[queue[i]] = true;
  var index = 0;
  while (index < queue.length) {
    var rowId = queue[index++];
    var linkedValue = "";
    try { linkedValue = sheet.getRange(rowId, 28).getValue().toString(); } catch(e) {}
    var linkedRows = parseLinkedRowIds(linkedValue, sheetName);
    for (var j = 0; j < linkedRows.length; j++) {
      if (!seen[linkedRows[j]]) {
        seen[linkedRows[j]] = true;
        queue.push(linkedRows[j]);
      }
    }
  }
  queue.sort(function(a, b) { return a - b; });
  return queue;
}

function writeLinkedGroup(sheet, rowIds) {
  var ids = parseLinkedRowIds(rowIds, "");
  if (ids.length === 0) return;
  var linkedString = ids.join(",");
  for (var i = 0; i < ids.length; i++) sheet.getRange(ids[i], 28).setValue(linkedString);
}

function clearLinkedGroup(sheet, rowIds) {
  var ids = parseLinkedRowIds(rowIds, "");
  for (var i = 0; i < ids.length; i++) sheet.getRange(ids[i], 28).clearContent();
}

function relinkOrderRow(sheet, targetRow, linkedRows, sheetName) {
  var newGroup = parseLinkedRowIds(linkedRows, sheetName);
  if (newGroup.indexOf(targetRow) === -1) newGroup.push(targetRow);
  newGroup.sort(function(a, b) { return a - b; });

  var groups = [getCompleteLinkedRows(sheet, [targetRow], sheetName)];
  for (var i = 0; i < newGroup.length; i++) {
    groups.push(getCompleteLinkedRows(sheet, [newGroup[i]], sheetName));
  }

  var affected = [];
  for (var g = 0; g < groups.length; g++) {
    for (var j = 0; j < groups[g].length; j++) {
      if (affected.indexOf(groups[g][j]) === -1) affected.push(groups[g][j]);
    }
  }

  clearLinkedGroup(sheet, affected);
  for (var k = 0; k < groups.length; k++) {
    var remaining = groups[k].filter(function(id) { return newGroup.indexOf(id) === -1; });
    remaining.sort(function(a, b) { return a - b; });
    if (remaining.length > 1) writeLinkedGroup(sheet, remaining);
  }
  if (newGroup.length > 1) writeLinkedGroup(sheet, newGroup);
}

function getNextOrderRow(sheet) {
  var startRow = 6; //
  var lastRowInSheet = sheet.getLastRow(); //
  if (lastRowInSheet > MAX_ROW_LIMIT) lastRowInSheet = MAX_ROW_LIMIT; //
  var targetRow = startRow; //
  if (lastRowInSheet >= startRow) { //
    var columnValues = sheet.getRange(startRow, 5, lastRowInSheet - startRow + 1, 1).getValues(); //
    var maxIndex = -1; //
    for (var i = 0; i < columnValues.length; i++) { if (columnValues[i][0] !== "") maxIndex = i; //
    } //
    targetRow = startRow + maxIndex + 1; //
  } //
  return targetRow; //
}

function findRowByClientRequestId(sheet, clientRequestId) {
  if (!sheet || !clientRequestId) return -1;
  var startRow = 6;
  var lastRow = Math.min(sheet.getLastRow(), MAX_ROW_LIMIT);
  if (lastRow < startRow) return -1;
  var values = sheet.getRange(startRow, 30, lastRow - startRow + 1, 1).getValues();
  for (var i = 0; i < values.length; i++) {
    if (values[i][0] && values[i][0].toString() === clientRequestId.toString()) {
      return startRow + i;
    }
  }
  return -1;
}

function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet(); //
    var activeSheetName = getActiveSheetName(); //
    var allData = []; //
    var monthlyStats = {}; //
    var allSheetNames = []; //
    
    var params = e && e.parameter ? e.parameter : {}; //
    var earlyBoxLinkResponse = handleBoxLinkAction(params);
    if (earlyBoxLinkResponse) return earlyBoxLinkResponse;
    // --- 🔔 NOTIFICATIONS SYSTEM (GET) ---
    if (params.action === "get_notifications") { //
        var notifSheet = ss.getSheetByName("Notifications"); //
        var notifications = []; //
        if (notifSheet) { //
            var lastRow = notifSheet.getLastRow(); //
            if (lastRow > 1) { //
                var startRow = Math.max(2, lastRow - 50); //
                var dataRows = notifSheet.getRange(startRow, 1, lastRow - startRow + 1, 12).getValues(); //
                for (var i = 0; i < dataRows.length; i++) { //
                    var r = dataRows[i]; //
                    var readByArr = []; //
                    try { readByArr = JSON.parse(r[8] || "[]"); //
                    } catch(e) {} //
                    notifications.push({
                        id: r[0], type: r[1], message: r[2], senderRole: r[3], 
                        targetRoles: (r[4] || "").toString().split(","), orderId: r[5], orderSheet: r[6], 
                        timestamp: r[7], readBy: readByArr, isWarning: r[9] === "true", //
                        isFixed: r[10] === "true", fixedBy: r[11] //
                    });
                } //
            }
        }
        return ContentService.createTextOutput(JSON.stringify({ status: "success", notifications: notifications.reverse() })).setMimeType(ContentService.MimeType.JSON); //
    }

    if (params.action === "get_orders") {
      return ContentService.createTextOutput(JSON.stringify({ status: "success", orders: readOrdersLog(ss) })).setMimeType(ContentService.MimeType.JSON);
    }

    if (params.action === "get_gift_cards") {
      return ContentService.createTextOutput(JSON.stringify({ status: "success", gift_cards: readGiftCards(ss) })).setMimeType(ContentService.MimeType.JSON);
    }
    
    var requestedMonth = params.month; //
    var loadAllLocal = params.load_all_local === 'true'; //
    var loadActiveOnly = !requestedMonth && !loadAllLocal; //

    var orderSheetNames = getOrderSheetNames();
    for (var s = 0; s < orderSheetNames.length; s++) { // 
      var sName = orderSheetNames[s]; // 
      var sheet = ss.getSheetByName(sName); //
      if (sheet) { //
        if (!monthlyStats[sName]) { //
            allSheetNames.push(sName); //
            var rev=0, bal=0, cBuy=0, cWgt=0, cEtc=0, lost=0; // 
            try {
                 rev = sheet.getRange("A1").getValue();
                 bal = sheet.getRange("L1").getValue();
                 cBuy = sheet.getRange("F1").getValue(); 
                 cWgt = sheet.getRange("H1").getValue(); 
                 cEtc = sheet.getRange("J1").getValue();
                 lost = sheet.getRange("K3").getValue();
            } catch(err) {}

            var sheetCount = 0;
            if (loadAllLocal || requestedMonth === sName || (loadActiveOnly && sName === activeSheetName)) {
               var sheetData = readSheetData(sheet, sName);
               allData = allData.concat(sheetData);
               sheetCount = sheetData.length;
            } else {
               var lastR = sheet.getLastRow();
               if (lastR >= 4) { sheetCount = lastR - 3; } else { sheetCount = 0; }
            }
            monthlyStats[sName] = { revenue: parseFloat(rev)||0, balance: parseFloat(bal)||0, buy: parseFloat(cBuy)||0, wgt: parseFloat(cWgt)||0, etc: parseFloat(cEtc)||0, lost: parseFloat(lost)||0, count: sheetCount };
        }
      }
    }
    
    var response = { status: "success", data: allData, meta: { active_sheet: activeSheetName, monthly_stats: monthlyStats, all_months: allSheetNames } }; // 
    return ContentService.createTextOutput(JSON.stringify(response)).setMimeType(ContentService.MimeType.JSON); //
  } catch(error) { //
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })).setMimeType(ContentService.MimeType.JSON); //
  } //
}

function sumRange(sheet, rangeA1) {
  try {
    var values = sheet.getRange(rangeA1).getValues()[0]; //
    var total = 0; //
    for(var i=0; i<values.length; i++) total += (parseFloat(values[i]) || 0); //
    return total; //
  } catch(e) { return 0; //
  } //
}

function readOrdersLog(ss) {
  var logSheet = ss.getSheetByName("Orders");
  if (!logSheet || logSheet.getLastRow() < 2) return [];
  var lastRow = logSheet.getLastRow();
  var lastCol = logSheet.getLastColumn();
  var headers = logSheet.getRange(1, 1, 1, lastCol).getValues()[0];
  var values = logSheet.getRange(2, 1, lastRow - 1, lastCol).getValues();
  var orders = [];
  for (var i = 0; i < values.length; i++) {
    var rowObj = {};
    for (var c = 0; c < headers.length; c++) {
      if (headers[c]) rowObj[headers[c]] = values[i][c];
    }
    orders.push(rowObj);
  }
  return orders;
}

function getOrCreateOrdersLogSheet(ss) {
  var logSheet = ss.getSheetByName("Orders");
  if (!logSheet) logSheet = ss.insertSheet("Orders");
  var headers = [
    "Timestamp", "UniqueOrderID", "ProcessedSheet", "ProcessedRow", "Insta",
    "Name", "Place", "Phone", "Phone2", "Fib", "RawItems", "RawLink",
    "RawUnadjustedPrice", "InitialPayment", "Note", "Extra", "BoxName",
    "PrimaryURLs", "ProofURLs", "LinkedOrderIds", "ClientRequestID", "SubmittedBy", "SubmittedRole", "RawPayload"
  ];
  var currentHeader = logSheet.getRange(1, 1, 1, headers.length).getValues()[0];
  var needsHeader = true;
  for (var i = 0; i < currentHeader.length; i++) {
    if (currentHeader[i]) { needsHeader = false; break; }
  }
  if (needsHeader || currentHeader[headers.length - 1] !== headers[headers.length - 1]) {
    logSheet.getRange(1, 1, 1, headers.length).setValues([headers]);
    logSheet.setFrozenRows(1);
  }
  return logSheet;
}

function readSheetData(sheet, sheetName) {
  if (!sheet) { return []; } //

  var lastRow = sheet.getLastRow(); //
  if (lastRow > MAX_ROW_LIMIT) lastRow = MAX_ROW_LIMIT; //
  if (lastRow < 4) return []; //
  // Read all the way up to AB (Column 28)
  var dataRange = sheet.getRange(4, 1, lastRow - 3, 33); //
  var values = dataRange.getValues(); //
  var formulas = dataRange.getFormulas(); //
  var rowLinks = {};
  for (var linkScan = 0; linkScan < values.length; linkScan++) {
    var scanRowIndex = linkScan + 4;
    rowLinks[scanRowIndex] = parseLinkedRowIds(values[linkScan][27], sheetName);
  }

  function collectLinkedGroupForRead(seedRow) {
    var queue = [seedRow];
    var seen = {};
    seen[seedRow] = true;
    var pos = 0;
    while (pos < queue.length) {
      var current = queue[pos++];
      var links = rowLinks[current] || [];
      for (var li = 0; li < links.length; li++) {
        if (!seen[links[li]]) {
          seen[links[li]] = true;
          queue.push(links[li]);
        }
      }
      for (var otherRow in rowLinks) {
        if (rowLinks[otherRow].indexOf(current) > -1 && !seen[otherRow]) {
          seen[otherRow] = true;
          queue.push(parseInt(otherRow, 10));
        }
      }
    }
    queue.sort(function(a, b) { return a - b; });
    return queue;
  }
  
  var jsonData = []; //
  for (var i = 0; i < values.length; i++) { //
    var row = values[i]; //
    var rowIndex = i + 4; //
    
    var noteVal = row[17] ? row[17].toString() : ""; //
    var extraValue = row[18] ? row[18].toString() : ""; //
    var orderNoValue = extraValue !== "" ? extraValue : "Pending"; //
    // 🔗 READ LINKED ORDER IDs FROM COLUMN AB (Index 27)
    var linkedIds = collectLinkedGroupForRead(rowIndex).filter(function(id) { return id !== rowIndex; }); //

    var dateVal = "Unknown Date"; //
    var rawDate = row[16]; //
    if (rawDate) { //
        try { var d = new Date(rawDate); //
        if (!isNaN(d.getTime())) { dateVal = Utilities.formatDate(d, Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss"); } else { dateVal = rawDate.toString(); //
        } //
        } catch(e) { dateVal = rawDate.toString(); //
        } //
    }

    // PRIMARY PICTURE: ONLY V Formula (21)
    var primaryImgUrl = ""; //
    if (formulas[i][21] && formulas[i][21].indexOf("http") > -1) { //
        var m = formulas[i][21].match(/"(https?:\/\/[^"]+)"/); //
        if (m) primaryImgUrl = m[1]; //
    }

    // PROOF PICTURE: X (23) or W Formula (22)
    var proofImgUrl = row[23] ? row[23].toString() : ""; //
    if (!proofImgUrl && formulas[i][22] && formulas[i][22].indexOf("http") > -1) { //
        var m1 = formulas[i][22].match(/"(https?:\/\/[^"]+)"/); //
        if (m1) proofImgUrl = m1[1]; //
    }
    
    // WARNING PICTURE: Y Formula (24)
    var warningImgUrl = ""; //
    if (formulas[i][24] && formulas[i][24].indexOf("http") > -1) { //
        var m2 = formulas[i][24].match(/"(https?:\/\/[^"]+)"/); //
        if (m2) warningImgUrl = m2[1]; //
    }

    var proofUrlsArray = proofImgUrl ? proofImgUrl.split(",") : []; //

    if (row[1] || row[8]) { //
      jsonData.push({ //
        insta: row[0], name: row[1], place: row[2], fib: row[3], price: row[4], //
        box_cost: row[5], pics_text: row[6] != null ? row[6].toString() : "", shipping_cost: row[7], phone: row[8] != null ? row[8].toString() : "", //
        box_name: row[9] != null ? row[9].toString() : "", lost: row[10], profit: row[11], //
        extra: extraValue, note: noteVal, link: (row[15] || ""), initial_payment: row[13], //
        date: dateVal, orderNo: orderNoValue, trackNo: (row[12] || ""), //
        
        warningBase64: warningImgUrl, //
        image_url: proofImgUrl, //
        proof_urls: proofUrlsArray, //
        
        primary_urls: primaryImgUrl, //
        unique_order_id: row[30] || "",
        admin_name: row[31] || "",
        admin_role: row[32] || "",
        id: rowIndex, sheet_name: sheetName, //
        linkedOrderIds: linkedIds //
      }); //
    }
  }
  return jsonData; //
}

function handleOrder(data, sheetName) {
  var ss = SpreadsheetApp.getActiveSpreadsheet(); //
  var sheet = ss.getSheetByName(sheetName); //
  if (!sheet) { try { sheet = ss.insertSheet(sheetName); } catch(e) { return { 'result': 'error', 'message': 'Could not create sheet: ' + sheetName }; //
  } } //
  
  var targetRow; //
  var isUpdate = false; //
  var isLinkedNewOrder = false; //
  var baseLinkedRow = -1; //
  var incomingLinkedRows = parseLinkedRowIds(data.linkedOrderIds || [], sheetName); //
  var incomingLinkedRefs = parseOrderLinkRefs(data.linkedOrderIds || [], sheetName); //
  var existingRequestRow = (!data.row_id && data.client_request_id) ? findRowByClientRequestId(sheet, data.client_request_id) : -1; //
  if (data.row_id && data.row_id != "") { //
    targetRow = parseInt(data.row_id); //
    isUpdate = true; //
  } else if (existingRequestRow > -1) {
    targetRow = existingRequestRow;
    isUpdate = true;
  } else if (incomingLinkedRefs.length > 0) { //
    isLinkedNewOrder = true; //
    baseLinkedRow = incomingLinkedRefs[0].row; //
    targetRow = getNextOrderRow(sheet); //
  } else {
    targetRow = getNextOrderRow(sheet); //
  }
  if (targetRow > MAX_ROW_LIMIT) return { 'result': 'error', 'message': 'Sheet Full!' }; //
  var uniqueOrderId = data.unique_order_id || data.order_uid || data.master_order_id || "";
  if (isUpdate && !uniqueOrderId) uniqueOrderId = sheet.getRange(targetRow, 31).getValue().toString();
  if (!uniqueOrderId) uniqueOrderId = Utilities.getUuid();
  data.unique_order_id = uniqueOrderId;
  if (!isUpdate) appendRawOrderLog(ss, data, sheetName, targetRow, uniqueOrderId);

  data.name = transliterateToLatini(data.name); //
  data.place = transliterateToLatini(data.place); //
  data.pics_text = transliterateToLatini(data.pics_text); //
  data.extra = transliterateToLatini(data.extra); //
  data.note = transliterateToLatini(data.note || data.customer_note || ""); //
  var rawLink = (data.link || "").toString().trim(); //
  if (rawLink.match(/^https?:\/\//)) { data.link = rawLink; //
  } //
  else if (rawLink.indexOf('.') > -1 && rawLink.indexOf(' ') === -1) { data.link = "https://" + rawLink; //
  } //
  else { var match = rawLink.match(/(https?:\/\/[^\s]+)/); if (match) data.link = match[0]; else data.link = rawLink; //
  }
  
  var p1 = (data.phone || '').toString().replace(/\s/g, ''); //
  if (p1.startsWith('+964')) p1 = '0' + p1.substring(4); //
  else if (p1.startsWith('00964')) p1 = '0' + p1.substring(5); //
  if (p1.startsWith('7')) p1 = '0' + p1; //
  var p2 = (data.phone2 || '').toString().replace(/\s/g, ''); //
  if (p2.startsWith('+964')) p2 = '0' + p2.substring(4); //
  else if (p2.startsWith('00964')) p2 = '0' + p2.substring(5); //
  if (p2.startsWith('7')) p2 = '0' + p2; //
  var finalPhone = p1 + (p2 ? " / " + p2 : ""); //
  
  var priceInput = data.customer_price || data['price_iqd'] || data['price (IQD)'] || data.price || "0"; //
  if (typeof priceInput === 'object') { priceInput = JSON.stringify(priceInput); //
  } //
  var rawPrice = priceInput.toString().replace(/,/g, '').replace(/[^0-9.]/g, '').trim(); //
  var basePrice = parseFloat(rawPrice) || 0; //
  
  var rawInitial = (data.initial_payment || "0").toString().replace(/,/g, '').replace(/[^0-9.]/g, ''); //
  var initialPayment = parseFloat(rawInitial) || 0; //
  var placeText = (data.place || "").toString().trim().toLowerCase(); //
  
  var isNoLocation = placeText === "no location" || placeText === "-"; //
  var isErbil = (placeText.includes("erbil") || placeText.includes("hawler") || placeText.includes("hewler")); //
  var isOutsideErbil = (placeText.includes("kurdistan") || placeText.includes("outside erbil")); //
  var isOutsideKurdistan = (placeText.includes("iraq") || placeText.includes("outside kurdistan")); //
  
  var actualShipping = 0; //
  if (isNoLocation) actualShipping = 0; //
  else if (isErbil) actualShipping = 3000; //
  else if (isOutsideErbil) actualShipping = 5000; //
  else if (isOutsideKurdistan) actualShipping = 6000; //
  else actualShipping = 5000; //

  var hasShippingDetails = false; //
  if (data.shipping_details === true || data.shipping_details === "true" || data.with_details === true || data.with_details === "true" || data.shipping === "with details") { //
    hasShippingDetails = true; //
  } else { //
    var kurdistanCities = ["erbil", "hawler", "hewler", "sulaymaniyah", "slemani", "duhok", "kirkuk", "halabja", "zakho", "soran", "kalar", "rania", "ranya"]; //
    var matchedCity = kurdistanCities.find(function(c) { return placeText.includes(c); }); //
    var cleanPlace = placeText.replace(/[^a-z0-9]/g, ""); //
    if (matchedCity && cleanPlace.length > matchedCity.length + 3) { hasShippingDetails = true; //
    } //
    else if (!matchedCity && cleanPlace.length > 5) { hasShippingDetails = true; //
    } //
  }

  var isNewIncomingOrder = !isUpdate; //
  var freeShippingText = ((data.extra || "") + " " + (data.note || "")).toString(); //
  var isFreeDelivery = basePrice > 118000 || /free\s*shipping/i.test(freeShippingText) || /\bfree\b/i.test(data.extra || ""); //
  if (isFreeDelivery) { //
    var currentExtra = (data.extra || "").toString(); //
    if (!/\bfree\b/i.test(currentExtra)) data.extra = data.extra ? (data.extra + " Free") : "Free"; //
  } //
  var postedSheetPrice = data.sheet_price !== undefined && data.sheet_price !== "" ? data.sheet_price : ""; //
  var postedSheetPriceNum = postedSheetPrice !== "" //
    ? parseFloat(postedSheetPrice.toString().replace(/,/g, '').replace(/[^0-9.-]/g, '')) //
    : NaN; //
  var sheetItemPrice = isNewIncomingOrder //
    ? (isFreeDelivery ? Math.max(basePrice - actualShipping, 0) : basePrice) //
    : (!isNaN(postedSheetPriceNum) ? postedSheetPriceNum : basePrice); //
  
  if (!isUpdate && !isLinkedNewOrder && hasShippingDetails) { //
    var l1Cell = sheet.getRange("L1"); //
    var currentL1Value = parseFloat(l1Cell.getValue()) || 0; //
    l1Cell.setValue(currentL1Value + 500); //
  }

  sheet.getRange(targetRow, 1).setValue(data.insta || ''); //
  sheet.getRange(targetRow, 2).setValue(data.name || ''); //
  sheet.getRange(targetRow, 3).setValue(data.place || ''); //
  sheet.getRange(targetRow, 4).setValue(data.fib || ''); //
  if (data.price === '') { sheet.getRange(targetRow, 5).clearContent(); //
  } else { sheet.getRange(targetRow, 5).setValue(sheetItemPrice).setNumberFormat("#,##0"); //
  } //
  
  sheet.getRange(targetRow, 7).setValue(data.pics_text || ''); //
  sheet.getRange(targetRow, 9).setValue(finalPhone); //
  if (data.box_name !== undefined) { sheet.getRange(targetRow, 10).setValue(data.box_name); //
  } //

  var extraText = (data.extra || "").toString().trim(); //
  var noteText = (data.note || "").toString().trim(); //
  extraText = extraText.replace(/^[\s\-\|\,]+|[\s\-\|\,]+$/g, ''); //
  noteText = noteText.replace(/^[\s\-\|\,]+|[\s\-\|\,]+$/g, ''); //
  
  sheet.getRange(targetRow, 18).setValue(noteText); //
  sheet.getRange(targetRow, 19).setValue(extraText); //
  sheet.getRange(targetRow, 16).setValue(data.link || ''); //
  sheet.hideColumns(20); //
  sheet.getRange(targetRow, 14).setValue(initialPayment).setNumberFormat("#,##0"); //
  if (!isUpdate || data.restore === "true") { //
      var now = new Date(); //
      var formattedDate = Utilities.formatDate(now, Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm:ss"); //
      sheet.getRange(targetRow, 17).setValue(formattedDate); //
  }
  if (data.client_request_id) {
    sheet.getRange(targetRow, 30).setValue(data.client_request_id);
    sheet.hideColumns(30);
  }
  sheet.getRange(targetRow, 31).setValue(uniqueOrderId);
  sheet.hideColumns(31);
  sheet.getRange(targetRow, 32).setValue(data.admin_name || data.submitted_by || "");
  sheet.getRange(targetRow, 33).setValue(data.admin_role || data.submitted_role || "");
  sheet.hideColumns(32, 2);

  data.orderNo = (data.extra || "Pending"); //
  var finalItemPrice = basePrice; //
  // LOGICAL LINKING -> SAVED TO COLUMN AB (28) NOW!
  if (isLinkedNewOrder) { //
    relinkOrderRefs(targetRow, sheetName, data.linkedOrderIds || []); //
    finalItemPrice = basePrice; //
  } else if (isUpdate && data.linkedOrderIds !== undefined) { //
    relinkOrderRefs(targetRow, sheetName, data.linkedOrderIds || []); //
  }

  // 👇 SAVING THE PRIMARY PICTURE (COL V and AA)
  if (data.primary_urls && data.primary_urls.length > 5) { //
      sheet.getRange(targetRow, 27).setValue(data.primary_urls); //
      // AA
      var pUrls = typeof data.primary_urls === 'string' ? data.primary_urls.split(',') : data.primary_urls; //
      sheet.getRange(targetRow, 22).setFormula('=IMAGE("' + pUrls[0] + '")'); // V
  }

  // 👇 SAVING THE PROOF PICTURE (COL W and X)
  var proofUrls = []; //
  if (data.proof_urls && data.proof_urls.length > 5) { //
    if (typeof data.proof_urls === 'string') { //
      proofUrls = data.proof_urls.split(','); //
    } else if (Array.isArray(data.proof_urls)) { //
      proofUrls = data.proof_urls; //
    } //
  }

    sheet.getRange(targetRow, 23).setFormula('=IMAGE("' + proofUrls[0] + '")'); //
    // W (23)
    sheet.getRange(targetRow, 24).setValue(proofUrls.join(",")); // X (24)
  } else {
    // Clear out cells if proof image is removed
    sheet.getRange(targetRow, 23).clearContent(); //
    // W (23)
    sheet.getRange(targetRow, 24).clearContent(); // X (24)
  }

  var shouldSendTelegram = (!isUpdate && !data.row_id && !data.status_color && !data.action); //
  if (shouldSendTelegram) { //
      try { 
           var receiptShipping = actualShipping; //
           if (!isNoLocation ) receiptShipping = 0; //
           if (!isErbil ) receiptShipping = 3000; //
           if (!isOutsideErbil ) receiptShipping = 5000; //
           if (!isOutsideKurdistan) receiptShipping = 6000; //
          
          var calculatedTotal = finalItemPrice + (isFreeDelivery ? 0 : receiptShipping); //

          var pdfId = sendTelegramReceipt(data, p1, p2, finalItemPrice, calculatedTotal, isFreeDelivery, initialPayment); //
          var ids = []; if (txtId) ids.push(txtId); if (pdfId) ids.push(pdfId); //

            var rowsForTelegramIds = isLinkedNewOrder && completeLinkedRows ? completeLinkedRows : [targetRow]; //
            for (var tgRowIdx = 0; tgRowIdx < rowsForTelegramIds.length; tgRowIdx++) { //

            } //
            sheet.hideColumns(26); //
          } //
      } catch (e) { console.log("Telegram Error: " + e); //
      } //
  }
  return { 'result': 'success', 'status': 'success', 'message': isUpdate ? //
  'Updated Row ' + targetRow : 'Created Row ' + targetRow, 'unique_order_id': uniqueOrderId }; //
} //

function setLinkedGroupMarker(sheet, rowId) {
  var extraRange = sheet.getRange(rowId, 19);
  var extra = extraRange.getValue().toString();
  if (!/\bfree\b/i.test(extra)) extra = (extra + (extra ? " " : "") + "Free").trim();
  if (!/\[GROUP_SHIP_DEDUCTED\]/i.test(extra)) extra = (extra + (extra ? " " : "") + "[GROUP_SHIP_DEDUCTED]").trim();
  extraRange.setValue(extra);
}

function clearLinkedGroupMarker(sheet, rowId) {
  var extraRange = sheet.getRange(rowId, 19);
  extraRange.setValue(extraRange.getValue().toString().replace(/\s*\[GROUP_SHIP_DEDUCTED\]/ig, "").trim());
}

function getLinkedGroupDisplayPrice(sheet, rowId) {
  var price = getLinkedGroupPrice(sheet, rowId);
  var shipping = getLinkedGroupShipping(sheet, rowId);
  return (hasLinkedGroupFreeText(sheet, rowId) || price + shipping > 118000) ? price + shipping : price;
}

function applyLinkedGroupShipping(sheet, rowIds) {
  if (!rowIds || rowIds.length === 0) return;
  var shippingTotal = 0;
  for (var i = 0; i < rowIds.length; i++) {
    shippingTotal += parseFloat(String(sheet.getRange(rowIds[i], 8).getValue()).replace(/[^0-9.-]+/g, "")) || 0;
  }
  if (shippingTotal === 0 && rowIds.length > 1) shippingTotal = getLinkedGroupShipping(sheet, rowIds[0]);

  if (rowIds.length < 2) {
    clearLinkedGroupMarker(sheet, rowIds[0]);
    return;
  }

  for (var r = 0; r < rowIds.length; r++) {
    clearLinkedGroupMarker(sheet, rowIds[r]);
    var shippingCell = sheet.getRange(rowIds[r], 8);
    if (r === 0) {
      if (shippingTotal) shippingCell.setValue(shippingTotal).setNumberFormat("#,##0");
      else shippingCell.clearContent();
    } else {
      shippingCell.clearContent();
    }
  }
}

function mergeBuyingPrice(sheetName, rowIds, price, shipping) {
  var ss = SpreadsheetApp.getActiveSpreadsheet(); //
  var sheet = ss.getSheetByName(sheetName); //
  if (!sheet) return { 'result': 'error', 'message': 'Sheet not found' }; //
  if (!rowIds || rowIds.length === 0) return { 'result': 'error', 'message': 'No rows selected' }; //
  rowIds = parseLinkedRowIds(rowIds, sheetName); //
  if (rowIds.length === 0) return { 'result': 'error', 'message': 'No valid rows selected' }; //

  rowIds.sort(function(a, b) { return a - b; }); //
  try { //
    if (rowIds.length === 1) { //
      applyLinkedGroupShipping(sheet, rowIds); //
      relinkOrderRow(sheet, rowIds[0], [], sheetName); //
      return { 'result': 'success', 'status': 'success', 'message': 'Order unlinked successfully.' }; //
    }

    relinkOrderRow(sheet, rowIds[0], rowIds.slice(1), sheetName); //
    applyLinkedGroupShipping(sheet, rowIds); //
    return { 'result': 'success', 'status': 'success', 'message': 'Logically linked ' + rowIds.length + ' rows successfully.' //
    }; //
  } catch (e) { return { 'result': 'error', 'message': e.toString() }; //
  } //
}

function deleteOrder(rowId, sheetName, linkedOrderIds) {
  var ss = SpreadsheetApp.getActiveSpreadsheet(); //
  var sheet = ss.getSheetByName(sheetName); //
  
  if (!sheet) { //
    for (var x = 0; x < EXTERNAL_SOURCES.length; x++) { //
      if (EXTERNAL_SOURCES[x].name === sheetName) { //
        try {
          var extSS = SpreadsheetApp.openById(EXTERNAL_SOURCES[x].id); //
          sheet = extSS.getSheetByName(EXTERNAL_SOURCES[x].tabName); //
        } catch(e) {}
        break; //
      } //
    }
  }

  if (!sheet) return { 'result': 'error', 'message': 'Sheet not found.' }; //
  var rId = parseInt(rowId); //
  if (rId > MAX_ROW_LIMIT) return { 'result': 'error', 'message': 'Safety Lock.' }; //

  var linkedRows = getCompleteLinkedRows(sheet, [rId], sheetName); //
  var hasLinkedSurvivors = linkedRows.filter(function(id) { return id !== rId; }).length > 0; //
  if (hasLinkedSurvivors) { //
    clearLinkedGroup(sheet, linkedRows); //
    var survivorsBeforeDelete = linkedRows.filter(function(id) { return id !== rId; }); //
    if (survivorsBeforeDelete.length > 1) writeLinkedGroup(sheet, survivorsBeforeDelete); //
  }

  if (!hasLinkedSurvivors) { //
    try { //
      var idString = sheet.getRange(rId, 26).getValue().toString(); //
      if (idString && idString !== "") { //
        var ids = idString.split(","); //
        for (var i = 0; i < ids.length; i++) { //
          if (ids[i].trim() !== "") deleteTelegramMessage(ids[i].trim()); //
        }
      }
    } catch(e) {}
  }

  sheet.getRange(rId, 1, 1, 33).clearContent().setBackground(null); //

  return { 'result': 'success', 'status': 'success', 'message': 'Row ' + rId + ' cleared.' }; //
}

function deleteFromSheetMenu() {
  var sheet = SpreadsheetApp.getActiveSheet(); //
  var row = sheet.getActiveCell().getRow(); //
  if (row < 4) { //
    SpreadsheetApp.getUi().alert("Please select a valid order row (Row 4 or below)."); //
    return; //
  } //
  
  var ui = SpreadsheetApp.getUi(); //
  var response = ui.alert('Confirm Delete', 'Are you sure you want to delete Row ' + row + ' and its Telegram messages?', ui.ButtonSet.YES_NO); //
  if (response == ui.Button.YES) { //
    var idString = sheet.getRange(row, 26).getValue().toString(); //
    if (!idString || idString === "") { //
        ui.alert("⚠️ Error: No Telegram Message IDs found in Column Z for this row. The script doesn't know what to delete on Telegram."); //
        return; //
    }
    
    var ids = idString.split(","); //
    var log = []; //
    for (var i = 0; i < ids.length; i++) { //
      if (ids[i].trim() !== "") { //
        var status = deleteTelegramMessage(ids[i].trim()); //
        log.push("Message ID " + ids[i].trim() + " -> " + status); //
      } //
    }
    
    sheet.getRange(row, 1, 1, 33).clearContent().setBackground(null); //
    ui.alert("Deletion Results:\n\n" + log.join("\n")); //
  }
}
